\subsection{Unperturbed chiral pairing}

Let
\[
D_0=
\begin{pmatrix}
0 & A^\ast\\
A & 0
\end{pmatrix}
\]
be the unperturbed $\mathbb Z_2$-graded Dirac operator on the relevant homogeneous
spinor bundle over
\[
M=G_2/SO(4),
\qquad \dim M=8,
\]
and set
\[
H_+^{(0)}=A^\ast A,
\qquad
H_-^{(0)}=AA^\ast.
\]

\begin{proposition}[Positive spectral pairing]
\label{prop:positive-pairing}
The non-zero spectra of $H_+^{(0)}$ and $H_-^{(0)}$ agree including
multiplicities. Consequently, after removing zero modes,
\[
K_+^{(0),\ast}(t)=K_-^{(0),\ast}(t)
\qquad (t>0),
\]
and therefore
\[
\zeta_+^{(0)}(s)=\zeta_-^{(0)}(s)
\]
wherever the zeta functions are initially defined, hence by meromorphic
continuation on their common domain. In particular,
\[
\Delta\zeta^{(0)\prime}(0)=0.
\]
\end{proposition}

\begin{proof}
For every $\lambda>0$, the map
\[
A:\ker(A^\ast A-\lambda)\longrightarrow\ker(AA^\ast-\lambda)
\]
is an isomorphism with inverse $\lambda^{-1}A^\ast$. Thus the positive
spectra agree with multiplicity. The identities for the reduced heat traces
and spectral zeta functions follow term by term.
\end{proof}

\subsection{Sector-dependent Fano--Chern--Weil deformation}
